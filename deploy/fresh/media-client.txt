// Progress is XHR request-body transport only, never durable publication.
const freshMediaRequests = new WeakMap();
function videoRetryIsVideo(url) { return /\.(mp4|mov|webm)(?:$|\?)/.test(url); }
function uploadMedia(blob, onProgress) {
  const actor = selectedPerson?.token || "";
  let record = freshMediaRequests.get(blob);
  if (!record || record.actor !== actor) {
    record = { key: crypto.randomUUID(), actor };
    freshMediaRequests.set(blob, record);
  }
  return new Promise((resolve, reject) => {
    const xhr = new XMLHttpRequest();
    xhr.open("POST", "/api/media");
    xhr.setRequestHeader("Content-Type", blob.type);
    xhr.setRequestHeader("X-Session", actor);
    xhr.setRequestHeader("Idempotency-Key", record.key);
    xhr.timeout = 210000; xhr.responseType = "json";
    xhr.upload.onprogress = event => {
      if (event.lengthComputable && onProgress) onProgress(event.loaded / event.total);
    };
    xhr.onload = () => {
      const data = xhr.response;
      if (xhr.status === 401) { clearSession(); reject(new Error("employee")); return; }
      if (xhr.status === 413) { reject(new Error("big")); return; }
      if (xhr.status === 415) { reject(new Error(data?.error === "video_too_long" ? "media_duration" : "media_format")); return; }
      if (xhr.status === 409) { reject(new Error("post_conflict")); return; }
      if (xhr.status === 429) { reject(new Error("post_limited")); return; }
      if (data?.error === "media_provider_unconfigured") { reject(new Error("post_media")); return; }
      if (xhr.status < 200 || xhr.status >= 300 || !data?.url) { reject(new Error("media_uncertain")); return; }
      resolve(data.url);
    };
    xhr.onerror = xhr.ontimeout = xhr.onabort = () => reject(new Error("media_uncertain"));
    xhr.send(blob);
  });
}
