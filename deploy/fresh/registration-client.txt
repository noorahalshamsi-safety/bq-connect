const freshEmailInput = document.querySelector("#mobile");
freshEmailInput.type = "email";
freshEmailInput.autocomplete = "email";
freshEmailInput.inputMode = "email";
freshEmailInput.maxLength = 254;
freshEmailInput.removeAttribute("data-i18n-placeholder");
freshEmailInput.placeholder = "name@binquraya.com";
const freshEmailLabel = freshEmailInput.closest("label").querySelector("span");
freshEmailLabel.removeAttribute("data-i18n");
freshEmailLabel.textContent = "Company email / البريد الإلكتروني للشركة";
document.querySelector("#code").closest("label").hidden = true;
document.querySelector("#code").disabled = true;
const freshRegistrationNote = document.querySelector("#register-form .login-secure");
freshRegistrationNote.removeAttribute("data-i18n");
freshRegistrationNote.textContent = "Your company email must match your employee record. Authenticator is optional after registration. / يجب أن يطابق البريد سجل الموظف. تطبيق المصادقة اختياري بعد التسجيل.";
let freshRegistrationTicket = "";
document.querySelector("#register-form").addEventListener("submit", async (event) => {
  event.preventDefault();
  const error = document.querySelector("#register-error"), done = document.querySelector("#register-done");
  const button = document.querySelector("#register-btn");
  const badge = cleanBadgeInput(document.querySelector("#reg-badge").value);
  const password = document.querySelector("#reg-password").value;
  const email = freshEmailInput.value.trim();
  error.textContent = ""; done.hidden = true;
  if (!badge || !email || !passwordStrong(password)) { error.textContent = t("registerMissing"); return; }
  if (password !== document.querySelector("#reg-password2").value) { error.textContent = t("pwMismatch"); return; }
  button.disabled = true;
  try {
    const reply = await postJson("/api/register", { badge, email, password,
      emailCode: freshRegistrationTicket ? document.querySelector("#otp").value : "",
      ticket: freshRegistrationTicket, app: standaloneApp() });
    if (reply.ok && reply.data.needEmailCode) {
      freshRegistrationTicket = reply.data.ticket;
      document.querySelector("#otp-field").hidden = false;
      done.textContent = "Check your company email for the code / أدخل الرمز المرسل إلى بريدك";
      done.hidden = false; document.querySelector("#otp").focus(); return;
    }
    if (!reply.ok || !reply.data.registered) {
      error.textContent = reply.data.error === "email_ownership_unverified" ?
        "Email-to-badge ownership is not verified. Contact your administrator. / لم يتم إثبات ربط البريد بالرقم الوظيفي." :
        t(reply.status === 429 ? "limited" : REGISTER_ERRORS[reply.data.error] || "loginFailed");
      return;
    }
    freshRegistrationTicket = "";
    ["#reg-badge", "#mobile", "#otp", "#reg-password", "#reg-password2"].forEach(id => document.querySelector(id).value = "");
    document.querySelector("#otp-field").hidden = true;
    loginStep(reply.data);
  } catch { error.textContent = t("loginFailed"); }
  finally { button.disabled = false; }
});
for (const id of ["#reg-badge", "#mobile"]) document.querySelector(id).addEventListener("input", () => {
  freshRegistrationTicket = ""; document.querySelector("#otp").value = "";
  document.querySelector("#otp-field").hidden = true;
});
